import asyncio
import json
from collections import deque
from bleak import BleakScanner

TARGET_NAME = "BCPro_217898"
WINDOW_SIZE = 10

readings = deque(maxlen=WINDOW_SIZE)


def get_status(avg_rssi):
    if avg_rssi >= -62:
        return "Very Near"
    elif avg_rssi >= -76:
        return "Near"
    elif avg_rssi >= -86:
        return "Far"
    else:
        return "Very Far / Weak"


def advertisement_received(device, advertisement_data):
    name = advertisement_data.local_name or device.name

    if name == TARGET_NAME:
        rssi = advertisement_data.rssi
        readings.append(rssi)

        avg_rssi = sum(readings) / len(readings)
        status = get_status(avg_rssi)

        output = {
            "name": TARGET_NAME,
            "rssi": rssi,
            "average_rssi": round(avg_rssi, 1),
            "status": status
        }

        print(json.dumps(output), flush=True)


async def main():
    scanner = BleakScanner(advertisement_received)

    await scanner.start()

    try:
        await asyncio.Event().wait()
    finally:
        await scanner.stop()


if __name__ == "__main__":
    asyncio.run(main())
